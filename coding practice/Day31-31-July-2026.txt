resource "datadog_monitor" "high_cpu" {
  name              = "High CPU Usage"
  type              = "metric alert"
  query             = "avg(last_5m):avg:system.cpu.user{env:${var.env}} > 80"
  message           = "CPU is high on {{host.name}}"
}

terraform {
  required_providers {
    datadog = {
      source = "DataDog/datadog"
    }
  }
  required_version = ">= 1.1.5"
}

  name = "Low Memory"
  type = "metric alert"

  query = "avg(last_5m):avg:system.mem.pct_usable{env:${var.env}} < 0.2"

  message = <<EOM

Check running processes and cleanup memory usage.
EOM

  tags = ["env:${var.env}", "team:devops"]
