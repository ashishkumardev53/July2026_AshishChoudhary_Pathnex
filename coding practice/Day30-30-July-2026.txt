--This is Ansible code 

---
- name: Create User and Configure SSH
  hosts: all
  become: yes

  tasks:

    - name: Create User
      user:
        name: pathnex-admin
        shell: /bin/bash
        state: present

    - name: Create .ssh Directory
      file:
        path: /home/pathnex-admin/.ssh
        state: directory
        owner: pathnex-admin
        group: pathnex-admin
        mode: '0700'

    - name: Add SSH Key
      copy:
        dest: /home/pathnex-admin/.ssh/authorized_keys
        content: "ssh-rsa AAAAB3NzaC1yc2EAAAADAQABAAABAQCexamplekey"
        owner: pathnex-admin
        group: pathnex-admin
        mode: '0600'

1)
hostname -I

2)

--This is Jenkins code  

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM = "DevOps"
        ENV = "prod"
        PROJECT = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {
            steps {
                dir('terraform') {
                    sh 'terraform init'
                    sh 'terraform apply -auto-approve'
                }
            }
        }

        stage('App Deployment') {
            steps {
                dir('ansible') {
                    sh 'ansible-playbook -i inventory playbook.yml'
                }

                dir('kubernetes') {
                    sh 'kubectl apply -f k8s.yaml'
                }
            }
        }

        stage('Monitoring Checks') {
            steps {
                sh 'bash menu.sh'
            }
        }
    }

    post {
        success {
            echo 'Pipeline Success'
        }

        failure {
            echo 'Pipeline Failed'
        }
    }
}
